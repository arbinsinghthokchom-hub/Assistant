// Injected variables:
// startId, endId, folderName, clanLink, waitTimeLeaveMs, finalSleepMs

var pkg = "com.supercell.clashofclans";
var dataPath = "/data/data/" + pkg;
var baseDir = "/sdcard/" + folderName + "/";

// Join Button Rectangle Area
var joinRect = {
    x1: 930,
    y1: 420,
    x2: 1230,
    y2: 570
};

log(">>> SCRIPT START: Req & Leave Specific (Advanced)");

/**
 * Loads a template from the Firebase Online system.
 * Returns the Image object or null if not downloaded.
 */
function readOnlineTemplate(name) {
    // 1. Use Packages prefix to force Rhino to search the root classloader
    // 2. Use .INSTANCE to explicitly target the Kotlin singleton if static bridge fails
    try {
        var Manager = Packages.com.example.autoreq.OnlineTemplateManager;
        var path = Manager.getTemplatePath(name);
        
        if (path != null) {
            log(">>> Loading Online Template: " + name);
            return images.read(path);
        }
    } catch (e) {
        log(">>> OnlineTemplateManager Error: " + e.message);
    }
    
    log(">>> Online Template NOT FOUND: " + name);
    return null;
}

/*
Performs a click using Accessibility
with random offset and duration.
*/
function humanClick(x, y, isFixed) {

    var finalX = x;
    var finalY = y;

    if (!isFixed) {
        // Random offset between -3 and +3
        var offsetX = Math.floor(Math.random() * 7) - 3;
        var offsetY = Math.floor(Math.random() * 7) - 3;

        finalX += offsetX;
        finalY += offsetY;
    }

    // Random press duration: 50ms - 70ms
    var duration = Math.floor(Math.random() * 21) + 50;

    // Accessibility press
    press(finalX, finalY, duration);

    // Post click delay: 250ms - 260ms
    var postDelay = Math.floor(Math.random() * 11) + 250;
    sleep(postDelay);
}

function checkPaused() {
    while (isPaused()) {
        status("PAUSED");
        sleep(2000);
    }
}

for (var i = startId; i <= endId; i++) {

    checkPaused();

    status("ACCOUNT : " + i);

    // ================= FORCE STOP =================

    shell("am force-stop " + pkg, true);
    shell("killall -9 " + pkg, true);

    sleep(500);

    // ================= SWITCH ACCOUNT =================

    var sourceFolder = baseDir + i + "/shared_prefs";

    var uidResult = shell("stat -c %u " + dataPath, true).result;
    var uid = uidResult ? uidResult.trim() : "";

    if (!uid) {
        uidResult = shell(
            "ls -ld " + dataPath + " | awk '{print $3}'",
            true
        ).result;

        uid = uidResult ? uidResult.trim() : "";
    }

    if (!uid) {
        log(">>> UID NOT FOUND FOR ID " + i);
        continue;
    }

    shell("rm -rf " + dataPath + "/shared_prefs", true);

    var cp = shell(
        "cp -af \"" + sourceFolder + "\" " + dataPath + "/",
        true
    );

    if (cp.code !== 0) {
        log(">>> COPY FAILED FOR ID " + i);
        continue;
    }

    // ================= FIX PERMISSIONS =================

    shell(
        "chown -R " + uid + ":" + uid + " " +
        dataPath + "/shared_prefs",
        true
    );

    shell(
        "chmod -R 777 " +
        dataPath + "/shared_prefs",
        true
    );

    shell(
        "chcon -R u:object_r:app_data_file:s0 " +
        dataPath + "/shared_prefs",
        true
    );

    // ================= OPEN CLAN LINK =================

    log(">>> Launching ID " + i + " via Clan Link");

    shell(
        "am start -a android.intent.action.VIEW -d \"" +
        clanLink + "\" " + pkg,
        true
    );

    status("LOADING...");

    // Decreased initial wait from 12000 to 10000 ms
    sleep(10000);

    // ================= DETECT JOIN AREA =================

    var foundJoin = false;
    var detectStart = new Date().getTime();
    
    // Load templates from Firebase Storage via the new system
    var consentImg = readOnlineTemplate("consent_popup.png");
    var joinImg = readOnlineTemplate("join.png");

    // Requirement: Caller decides what to do if null
    if (!joinImg) {
        status("MISSING ONLINE TEMPLATE");
        log(">>> Critical Error: join.png not found in Firebase local storage.");
        // Optionally: fallback to images.read("join.png") or skip ID
        continue; 
    }

    while ((new Date().getTime() - detectStart) < 35000) {

        checkPaused();

        // Use the native Kotlin Service Capture Method
        var MyAuto = com.example.autoreq.MyAutoService;
        var s = MyAuto.instance;
        var bitmap = s ? s.captureScreenSync() : null;

        if (bitmap) {
            // Convert Bitmap to Auto.js ImageWrapper
            var screen = com.stardust.autojs.core.image.ImageWrapper.ofBitmap(bitmap);

            // 1. Check for Consent Popup (Safely wrapped in case it didn't download)
            if (consentImg) {
                var consentPos = findImage(screen, consentImg);
                if (consentPos) {
                    log(">>> Detected consent_popup.png");
                    humanClick(900, 560, false);
                    sleep(300);
                    foundJoin = true;
                    screen.recycle(); // Free memory
                    break;
                }
            }

            // 2. Check for Join Button
            var joinPos = findImage(screen, joinImg);
            if (joinPos) {
                log(">>> Detected join.png");
                foundJoin = true;
                screen.recycle(); // Free memory
                break;
            }
            
            screen.recycle(); // Free memory for failed loop iterations
        } else {
            log("WARN: Capture failed or returned null");
        }

        sleep(1500);
    }
    
    // Free images from memory to prevent OutOfMemory errors in long loops
    if (consentImg) consentImg.recycle();
    if (joinImg) joinImg.recycle();

    // ================= JOIN + REQUEST =================

    if (foundJoin) {

        // JOIN BUTTON
        humanClick(1079, 495, false);

        // Wait 200-250ms
        var okWait = Math.floor(Math.random() * 51) + 200;

        sleep(okWait);

        // OK BUTTON
        humanClick(793, 488, false);

        sleep(2000);

        // ================= REQUEST SEQUENCE =================

        status("REQUESTING...");

        humanClick(1240, 44, false);   // coin
        humanClick(488, 528, false);   // cancel popup
        humanClick(56, 520, false);    // enter
        humanClick(979, 537, true);    // delete icon
        humanClick(864, 499, false);   // okay
        humanClick(1177, 574, false);  // request
        humanClick(852, 493, false);   // send
        humanClick(1260, 59, false);   // exit

        // ================= LEAVE SEQUENCE =================

        status("LEAVING...");

        humanClick(55, 330, false);    // message
        humanClick(245, 502, false);   // i understand
        humanClick(100, 39, false);    // online
       

        log(">>> Waiting " + (waitTimeLeaveMs / 1000) + "s before leave...");

        sleep(waitTimeLeaveMs);
        humanClick(1074, 491, false);  // leave
        // CONFIRM OK
        humanClick(789, 476, false);

        sleep(1000);

        status("ID " + i + " DONE");

    } else {

        status("TIMEOUT ID " + i);
    }

    // ================= FORCE STOP AGAIN =================

    shell("am force-stop " + pkg, true);

    sleep(500);
}

// ================= FINAL CYCLE =================

status("CYCLE DONE. SLEEPING...");

log(">>> Final Sleep: " + (finalSleepMs / 1000) + " seconds");
